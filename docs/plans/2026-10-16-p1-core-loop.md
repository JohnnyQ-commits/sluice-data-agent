```python
"""内置工具 query_data——demo SQLite 库只读查询，P1 数据周报的数据通道。"""

from __future__ import annotations

import sqlite3
from typing import Any

from .base import Tool, ToolResult

_MAX_ROWS = 50


class QueryDataTool(Tool):
    name = "query_data"
    description = "在 SQLite 数据库上执行只读 SQL（仅 SELECT/WITH），结果以 markdown 表格返回，最多 50 行。"
    parameters = {
        "type": "object",
        "properties": {
            "sql": {"type": "string", "description": "要执行的只读 SQL 查询"},
        },
        "required": ["sql"],
    }

    def __init__(self, db_path: str):
        self._db_path = db_path

    def execute(self, args: dict[str, Any]) -> ToolResult:
        sql = (args.get("sql") or "").strip()
        first = sql.split(None, 1)[0].lower() if sql else ""
        if first not in ("select", "with"):
            return ToolResult("query_data 仅接受只读查询（SELECT/WITH 开头）", is_error=True)
        try:
            conn = sqlite3.connect(f"file:{self._db_path}?mode=ro", uri=True)
            try:
                cur = conn.execute(sql)
                rows = cur.fetchmany(_MAX_ROWS + 1)
                cols = [d[0] for d in cur.description] if cur.description else []
            finally:
                conn.close()
        except sqlite3.Error as exc:
            return ToolResult(f"SQL 执行失败: {exc}", is_error=True)

        truncated = len(rows) > _MAX_ROWS
        rows = rows[:_MAX_ROWS]
        lines = ["| " + " | ".join(cols) + " |", "|" + "---|" * len(cols)]
        lines += ["| " + " | ".join("" if v is None else str(v) for v in r) + " |" for r in rows]
        if truncated:
            lines.append(f"（仅展示前 {_MAX_ROWS} 行）")
        return ToolResult("\n".join(lines))
```

```python
"""Skill 包协议与加载器（P1 最小面）。

Skill 包 = 子目录 + SKILL.md（frontmatter: name/description）+ 可选 tools.py（模块级 TOOLS 列表）。
私有工具以 {skill_name}.{tool_name} 前缀注册，模型视角仍在同一个扁平工具池。
skill.py 钩子与 evals/ 由 P2/P3 消费。
"""

from __future__ import annotations

import importlib.util
import re
import sys
from dataclasses import dataclass, field
from pathlib import Path

from ..tools.base import Tool
from ..tools.registry import ToolRegistry

_FRONTMATTER_RE = re.compile(r"\A---\s*\n(.*?)\n---", re.DOTALL)


def parse_skill_md(text: str) -> dict[str, str]:
    """解析 SKILL.md 顶部 YAML frontmatter（平铺 key: value）。"""
    m = _FRONTMATTER_RE.match(text)
    if not m:
        return {}
    meta: dict[str, str] = {}
    for line in m.group(1).splitlines():
        if ":" in line:
            key, _, value = line.partition(":")
            meta[key.strip()] = value.strip()
    return meta


@dataclass
class SkillPackage:
    name: str
    description: str
    path: Path
    tools: list[Tool] = field(default_factory=list)

    def prompt_line(self) -> str:
        return f"- {self.name}: {self.description}"


def _load_tools(py_path: Path, skill_name: str) -> list[Tool]:
    spec = importlib.util.spec_from_file_location(f"sluice_skill_{skill_name}_tools", py_path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return list(getattr(module, "TOOLS", []))


class SkillLoader:
    def load(self, skills_dir: str | Path, registry: ToolRegistry | None = None) -> list[SkillPackage]:
        root = Path(skills_dir)
        if not root.is_dir():
            return []
        packages: list[SkillPackage] = []
        for d in sorted(p for p in root.iterdir() if p.is_dir()):
            md = d / "SKILL.md"
            if not md.is_file():
                continue
            meta = parse_skill_md(md.read_text(encoding="utf-8"))
            if not meta.get("name"):
                continue
            tools = _load_tools(d / "tools.py", meta["name"]) if (d / "tools.py").is_file() else []
            for t in tools:
                t.name = f"{meta['name']}.{t.name}"
                if registry is not None:
                    registry.register(t)
            packages.append(
                SkillPackage(
                    name=meta["name"],
                    description=meta.get("description", ""),
                    path=d,
                    tools=tools,
                )
            )
        return packages
```

```python
"""数据周报 Skill 私有工具。"""

from __future__ import annotations

from typing import Any

from sluice.tools.base import Tool, ToolResult


class RenderReportTool(Tool):
    name = "render_report"
    description = "把指标渲染成 markdown 数据周报。"
    parameters = {
        "type": "object",
        "properties": {
            "title": {"type": "string", "description": "报告标题"},
            "period": {"type": "string", "description": "统计周期，如 2026-10-13 ~ 2026-10-19"},
            "metrics": {
                "type": "array",
                "description": "核心指标列表",
                "items": {
                    "type": "object",
                    "properties": {
                        "name": {"type": "string"},
                        "value": {"type": "string"},
                        "delta": {"type": "string", "description": "环比变化，如 +15%"},
                    },
                    "required": ["name", "value"],
                },
            },
            "highlights": {
                "type": "array",
                "items": {"type": "string"},
                "description": "重点发现列表",
            },
        },
        "required": ["title", "period", "metrics"],
    }

    def execute(self, args: dict[str, Any]) -> ToolResult:
        title = args.get("title")
        period = args.get("period")
        metrics = args.get("metrics")
        if not title or not period or not metrics:
            return ToolResult("title/period/metrics 均为必填", is_error=True)

        lines = [f"# {title}", "", f"**统计周期**：{period}", "", "## 核心指标", ""]
        for m in metrics:
            delta = m.get("delta")
            lines.append(f"- **{m['name']}**：{m['value']}" + (f"（环比 {delta}）" if delta else ""))
        highlights = args.get("highlights") or []
        if highlights:
            lines += ["", "## 重点发现", ""]
            lines += [f"- {h}" for h in highlights]
        lines += ["", "*本报告由 Sluice weekly-report Skill 生成*"]
        return ToolResult("\n".join(lines))
```

```python
"""Flask 壳——SSE 转发 + 静态托管，全部逻辑在框架层（设计文档 §7）。"""

from __future__ import annotations

import json
import queue
import threading
import uuid
from pathlib import Path

from flask import Flask, Response, jsonify, request, send_from_directory

from ..config.settings import Settings
from ..llm.client import LLMClient
from ..runtime.loop import ReActRuntime
from ..skills.loader import SkillLoader
from ..tools.query_data import QueryDataTool
from ..tools.registry import ToolRegistry

_FRONTEND_DIST = Path(__file__).resolve().parents[3] / "web" / "frontend" / "dist"


def create_app(client=None, registry=None, runtime=None) -> Flask:
    app = Flask(__name__, static_folder=None)
    settings = Settings()
    client = client or LLMClient(settings)
    if registry is None:
        registry = ToolRegistry()
        registry.register(QueryDataTool(settings.demo_db_path))
    packages = SkillLoader().load("skills", registry)
    runtime = runtime or ReActRuntime(
        client,
        registry,
        system=_build_system(packages),
        max_rounds=settings.react_max_rounds,
    )

    runs: dict[str, dict] = {}

    @app.post("/api/chat")
    def chat():
        body = request.get_json(silent=True) or {}
        if not body.get("message"):
            return jsonify({"error": "message 必填"}), 400
        run_id = uuid.uuid4().hex
        q: queue.Queue = queue.Queue()
        runs[run_id] = {"queue": q, "done": False}

        def worker():
            try:
                # on_event 覆盖构造器值（Task 3 契约）；runtime 自身会 emit final 事件，worker 不重复发
                runtime.run(body["message"], on_event=q.put)
            except Exception as exc:  # noqa: BLE001 — 异常作为 SSE 事件送达前端
                q.put({"type": "error", "message": str(exc)})
            finally:
                runs[run_id]["done"] = True

        threading.Thread(target=worker, daemon=True).start()
        return jsonify({"run_id": run_id})

    @app.get("/api/stream/<run_id>")
    def stream(run_id: str):
        run = runs.get(run_id)
        if run is None:
            return jsonify({"error": "未知 run_id"}), 404

        def generate():
            q = run["queue"]
            while True:
                try:
                    event = q.get(timeout=60)
                except queue.Empty:
                    yield 'event: end\ndata: {"type": "timeout"}\n\n'
                    return
                yield f"data: {json.dumps(event, ensure_ascii=False)}\n\n"
                if event.get("type") in ("final", "error"):
                    yield "event: end\ndata: {}\n\n"
                    return

        return Response(generate(), mimetype="text/event-stream")

    @app.get("/api/skills")
    def skills():
        return jsonify({"skills": [{"name": p.name, "description": p.description} for p in packages]})

    @app.get("/")
    def index():
        if (_FRONTEND_DIST / "index.html").is_file():
            return send_from_directory(_FRONTEND_DIST, "index.html")
        return "前端未构建：web/frontend/dist 不存在，先在 web/frontend 下 npm install && npm run build", 200

    @app.get("/assets/<path:filename>")
    def assets(filename: str):
        return send_from_directory(_FRONTEND_DIST / "assets", filename)

    return app


def _build_system(packages) -> str:
    base = "你是 Sluice 平台的助手。可以调用提供的工具完成任务，完成后直接给出最终回答。"
    if packages:
        lines = "\n".join(p.prompt_line() for p in packages)
        base += "\n\n可用的 Skill（通过其带前缀的工具使用）：\n" + lines
    return base
```

注意：`ReActRuntime.run` 在 Task 3 已带可选参数 `on_event`（非 None 时覆盖构造器值），worker 直接传 `q.put` 即可，final 事件由 runtime 统一 emit，worker 不重复发。

```jsx
import { useRef, useState } from "react";

const EVENT_LABEL = {
  round_start: "轮次",
  tool_call: "工具调用",
  observation: "观察结果",
  error: "错误",
};

function EventLine({ ev }) {
  if (ev.type === "round_start") {
    return <div className="mt-2 text-xs font-semibold text-gray-500">第 {ev.round} 轮</div>;
  }
  if (ev.type === "tool_call") {
    return (
      <div className="text-sm">
        <span className="font-mono text-indigo-600">{ev.name}</span>
        <code className="ml-2 rounded bg-gray-100 px-1 py-0.5 font-mono text-xs">
          {JSON.stringify(ev.args)}
        </code>
      </div>
    );
  }
  if (ev.type === "observation") {
    return (
      <details className="text-sm">
        <summary className={ev.is_error ? "text-red-600 cursor-pointer" : "text-gray-600 cursor-pointer"}>
          观察：{ev.name}
          {ev.is_error ? "（出错）" : ""}
        </summary>
        <pre className="mt-1 max-h-40 overflow-auto whitespace-pre-wrap rounded bg-gray-50 p-2 text-xs">
          {ev.observation}
        </pre>
      </details>
    );
  }
  if (ev.type === "error") {
    return (
      <div className="rounded border border-red-300 bg-red-50 p-2 text-sm text-red-700">
        运行出错：{ev.message}
      </div>
    );
  }
  return null;
}

export default function App() {
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState("");
  const [busy, setBusy] = useState(false);
  const bottomRef = useRef(null);

  async function send() {
    const text = input.trim();
    if (!text || busy) return;
    setInput("");
    setBusy(true);
    setMessages((m) => [...m, { role: "user", text }]);

    const runId = (
      await (
        await fetch("/api/chat", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ message: text }),
        })
      ).json()
    ).run_id;

    setMessages((m) => [...m, { role: "run", events: [] }]);
    const es = new EventSource(`/api/stream/${runId}`);
    es.onmessage = (e) => {
      const ev = JSON.parse(e.data);
      if (ev.type === "final") {
        es.close();
        setMessages((m) => {
          const copy = [...m];
          const last = copy[copy.length - 1];
          if (last && last.role === "run") last.answer = ev.answer;
          return copy;
        });
        setBusy(false);
      } else {
        setMessages((m) => {
          const copy = [...m];
          const last = copy[copy.length - 1];
          if (last && last.role === "run") last.events = [...last.events, ev];
          return copy;
        });
        bottomRef.current?.scrollIntoView({ behavior: "smooth" });
      }
    };
    es.onerror = () => {
      es.close();
      setBusy(false);
    };
  }

  return (
    <div className="mx-auto flex h-screen max-w-3xl flex-col p-4">
      <h1 className="mb-4 text-xl font-bold">Sluice — 可挂载 Skill 的数据 Agent 平台</h1>
      <div className="flex-1 space-y-3 overflow-y-auto rounded-lg border p-4">
        {messages.map((m, i) =>
          m.role === "user" ? (
            <div key={i} className="flex justify-end">
              <div className="max-w-[80%] rounded-2xl bg-blue-600 px-4 py-2 text-white">{m.text}</div>
            </div>
          ) : m.role === "run" ? (
            <div key={i} className="rounded-lg bg-gray-50 p-3">
              {m.events.map((ev, j) => (
                <EventLine key={j} ev={ev} />
              ))}
              {m.answer && (
                <pre className="mt-2 whitespace-pre-wrap rounded border bg-white p-3 font-sans text-sm">
                  {m.answer}
                </pre>
              )}
            </div>
          ) : null
        )}
        <div ref={bottomRef} />
      </div>
      <div className="mt-3 flex gap-2">
        <input
          className="flex-1 rounded-lg border px-3 py-2"
          value={input}
          placeholder={busy ? "Agent 运行中…" : "例如：生成上周的数据周报"}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === "Enter" && send()}
          disabled={busy}
        />
        <button className="rounded-lg bg-blue-600 px-4 py-2 text-white disabled:opacity-50" onClick={send} disabled={busy}>
          发送
        </button>
      </div>
    </div>
  );
}
```
