```python
"""评测层数据模型：用例 / 用例结果 / 报告（JSON 序列化 + 通过率 + judge 均分）。

EvalCase 由 Skill 包 evals/cases.py 的 CASES 列表项构造（from_dict 校验必填与未知键）。
三层评测共用同一套结果/报告形态：outcome（产出）/ behavior（单轮行为）/ guardrail（负向门禁）。
"""

from __future__ import annotations

import json
import time
from dataclasses import dataclass, field
from pathlib import Path

LAYERS = ("outcome", "behavior", "guardrail")

_ALLOWED_KEYS = {
    "name",
    "layer",
    "input",
    "must_contain",
    "judge_rubric",
    "expect_skill",
    "script",
    "expect_guardrails",
}


@dataclass
class EvalCase:
    skill: str
    name: str
    layer: str
    input: str
    must_contain: list[str] = field(default_factory=list)
    judge_rubric: list[str] = field(default_factory=list)
    expect_skill: str = ""
    script: list[dict] = field(default_factory=list)
    expect_guardrails: list[dict] = field(default_factory=list)

    @classmethod
    def from_dict(cls, skill: str, raw: dict) -> "EvalCase":
        unknown = set(raw) - _ALLOWED_KEYS
        if unknown:
            raise ValueError(f"评测用例含未知字段 {sorted(unknown)}（拼写错误？）: {raw!r}")
        for key in ("name", "layer", "input"):
            if not raw.get(key):
                raise ValueError(f"评测用例缺少必填字段 {key}: {raw!r}")
        if raw["layer"] not in LAYERS:
            raise ValueError(f"未知评测层 {raw['layer']!r}（应为 {'/'.join(LAYERS)}）")
        return cls(
            skill=skill,
            name=raw["name"],
            layer=raw["layer"],
            input=raw["input"],
            must_contain=list(raw.get("must_contain", [])),
            judge_rubric=list(raw.get("judge_rubric", [])),
            expect_skill=raw.get("expect_skill", ""),
            script=list(raw.get("script", [])),
            expect_guardrails=list(raw.get("expect_guardrails", [])),
        )


@dataclass
class CaseResult:
    skill: str
    case: str
    layer: str
    status: str  # pass | fail | error | skipped
    detail: dict = field(default_factory=dict)

    def to_dict(self) -> dict:
        return {
            "skill": self.skill,
            "case": self.case,
            "layer": self.layer,
            "status": self.status,
            "detail": self.detail,
        }


@dataclass
class EvalReport:
    results: list[CaseResult] = field(default_factory=list)
    generated_at: str = field(default_factory=lambda: time.strftime("%Y-%m-%d %H:%M:%S"))

    def pass_rate(self) -> float:
        """通过率；skipped 不进分母（无 API key 时 outcome/behavior 全 skipped，不算 0 分）。"""
        counted = [r for r in self.results if r.status != "skipped"]
        if not counted:
            return 1.0
        return sum(1 for r in counted if r.status == "pass") / len(counted)

    def judge_avg(self) -> float | None:
        """全体 outcome 用例的有效 judge 均分；无有效分返回 None。"""
        scores = [
            r.detail["judge_avg"]
            for r in self.results
            if r.layer == "outcome" and r.detail.get("judge_valid")
        ]
        if not scores:
            return None
        return sum(scores) / len(scores)

    def to_dict(self) -> dict:
        return {
            "generated_at": self.generated_at,
            "pass_rate": round(self.pass_rate(), 4),
            "judge_avg": self.judge_avg(),
            "results": [r.to_dict() for r in self.results],
        }

    def save(self, path: str | Path) -> None:
        p = Path(path)
        p.parent.mkdir(parents=True, exist_ok=True)
        p.write_text(json.dumps(self.to_dict(), ensure_ascii=False, indent=2), encoding="utf-8")

    @classmethod
    def load(cls, path: str | Path) -> "EvalReport":
        data = json.loads(Path(path).read_text(encoding="utf-8"))
        return cls(
            results=[CaseResult(**r) for r in data.get("results", [])],
            generated_at=data.get("generated_at", ""),
        )
```

    )
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return getattr(module, attr, None)

class BehaviorGate:
    """run 级状态（每次 run 新建实例）——跨 run 不共享记忆。

    重复判定两档：
    - 同轮内（同一条 assistant 消息）完全相同调用 → 打断（P3 补，堵同轮双发漏拦）
    - 与上一轮完全相同调用 → 打断；A→B→A 隔轮重复不算，留给 P3 evals 观察是否收紧
    check_call 放行即登记当轮指纹（claim-on-pass）：被拦调用不算已执行，
    同轮重试仍由输入门禁/重复判定接住，不会因 claim 缺失而漏拦。
    """

    def __init__(self, token_budget: int):
        self._budget = token_budget
        self._tokens_used = 0
        self._prev_keys: set[str] = set()
        self._round_keys: set[str] = set()

    def check_call(self, tu: ToolUseBlock) -> str | None:
        """工具执行前调用。返回 None 放行（并登记当轮指纹），返回打断原因（回喂模型）。"""
        key = call_key(tu)
        if key in self._round_keys:
            return (
                f"检测到同一轮内完全相同的重复调用 {tu.name}，已强制打断："
                "一次工具调用即可，请不要在同一条消息里重复发起"
            )
        if key in self._prev_keys:
            return (
                f"检测到与上一轮完全相同的重复调用 {tu.name}，已强制打断："
                "请换一种方法推进，或直接给出最终回答"
            )
        self._round_keys.add(key)
        return None

    def commit_round(self, tool_uses: list[ToolUseBlock]) -> None:
        """每轮末轮转比对基准：本轮全部指纹（含被拦的）成为下一轮基准，当轮登记清空。"""
        self._prev_keys = {call_key(tu) for tu in tool_uses}
        self._round_keys = set()

    def record_usage(self, usage: dict) -> str | None:
        """每轮末累计 token；超预算返回终止原因（loop 据此抛 BudgetExceededError）。"""

        )
        if self._budget and self._tokens_used > self._budget:
            return f"token 预算超限：已用 {self._tokens_used} > 预算 {self._budget}"
        return None

```python
"""三层评测执行器：guardrail（确定性）/ behavior（单轮）/ outcome（完整循环）。

评测引擎自身不依赖真实 LLM：runner 接受 client 参数，测试注入桩。
pytest 套件保持零 API key 可跑（P0 以来的红线）。
"""

from __future__ import annotations

from ..config.settings import Settings
from ..exceptions import SluiceError
from ..guardrails.chain import GuardrailChain
from ..llm.types import LLMResponse, TextBlock, ToolUseBlock
from ..runtime.loop import ReActRuntime
from ..tools.registry import ToolRegistry
from .model import CaseResult, EvalCase


def _response_from_spec(spec: dict) -> LLMResponse:
    """script 条目 → LLMResponse：{"tool","args"} → 工具调用；{"text"} → 最终回答。"""
    if "tool" in spec:
        return LLMResponse(
            content=[ToolUseBlock("t1", spec["tool"], spec.get("args", {}))],
            stop_reason="tool_use",
            usage={},
        )
    return LLMResponse(content=[TextBlock(spec.get("text", ""))], stop_reason="end_turn", usage={})


class ScriptedClient:
    """评测剧本回放（与 tests 同款模式，guardrail 层负向用例专用）。"""

    def __init__(self, responses: list[LLMResponse]):
        self._responses = list(responses)

    def chat(self, messages, system=None, tools=None):
        assert self._responses, "评测剧本耗尽：script 与运行时行为不符"
        return self._responses.pop(0)


class GuardrailRunner:
    """③ 负向门禁评测：剧本驱动完整 ReAct 循环，断言门禁事件真的发生。确定性、零 LLM 成本。"""

    def __init__(self, settings: Settings):
        self._settings = settings

    def run_case(self, case: EvalCase, registry: ToolRegistry) -> CaseResult:
        chain = GuardrailChain(self._settings, [])  # 无契约：此层只测拦截，不测产出
        client = ScriptedClient([_response_from_spec(s) for s in case.script])
        rt = ReActRuntime(client, registry, max_rounds=8, guardrails=chain)
        events: list[dict] = []
        try:
            rt.run(case.input, on_event=events.append)
        except SluiceError as exc:  # 轮次耗尽/超预算 = 剧本与行为不符或门禁终止
            return CaseResult(case.skill, case.name, case.layer, "error", {"error": str(exc)})
        got = [
            {"gate": e["gate"], "action": e["action"]}
            for e in events
            if e["type"] == "guardrail"
        ]
        missing = [x for x in case.expect_guardrails if x not in got]
        return CaseResult(
            case.skill,
            case.name,
            case.layer,
            "pass" if not missing else "fail",
            {"expected": case.expect_guardrails, "got": got},
        )
```

class OutcomeRunner:
    """① 产出评测：完整 ReAct 循环 + 确定性断言（must_contain）。

    语义断言（judge_rubric + LLM-as-judge）由 T7 接入本类。
    """

    def __init__(self, settings: Settings, client, judge_client=None):
        self._settings = settings
        self._client = client
        self._judge_client = judge_client

    def run_case(self, case: EvalCase, package, registry: ToolRegistry) -> CaseResult:
        chain = GuardrailChain(self._settings, [package])
        rt = ReActRuntime(
            self._client,
            registry,
            system=build_system([package]),  # build_system 已由 T5 加入模块 imports
            max_rounds=self._settings.react_max_rounds,
            guardrails=chain,
        )
        events: list[dict] = []
        try:
            answer = rt.run(case.input, on_event=events.append)
        except SluiceError as exc:
            return CaseResult(case.skill, case.name, case.layer, "fail", {"error": str(exc)})
        missing = [s for s in case.must_contain if s not in answer]
        detail: dict = {"missing": missing, "answer_head": answer[:200]}
        if case.judge_rubric and self._judge_client is not None:
            from .judge import judge_output  # T7 落地后生效；此前 judge_rubric 用例不带 judge 跑

            items = judge_output(self._judge_client, answer, case.judge_rubric)
            valid_scores = [i["score"] for i in items if i["valid"]]
            avg = sum(valid_scores) / len(valid_scores) if valid_scores else None
            detail["judge_items"] = items
            detail["judge_avg"] = avg
            detail["judge_valid"] = bool(valid_scores)
            if avg is not None and avg < self._settings.eval_judge_threshold:
                missing = missing or [f"judge 均分 {avg:.2f} < {self._settings.eval_judge_threshold}"]
        return CaseResult(
            case.skill,
            case.name,
            case.layer,
            "pass" if not missing else "fail",
            detail,
        )

"""LLM-as-judge：按 rubric 逐项打分，引用产出原文佐证——引用不到 = 判分无效（防作弊）。

judge 只看产出与 rubric，不给被评对象的 prompt/轨迹（设计文档 §3 防作弊约束）。
judge 挂掉/输出不合法 → 该项判分无效（score=None），用例退回纯确定性断言，不误伤。
"""

from __future__ import annotations

import json
import re

from ..llm.types import Message, TextBlock

JUDGE_SYSTEM = (
    "你是评测裁判。对给定产出按评分维度逐项打 1-5 分，"
    "每一项必须引用产出原文作为佐证。只输出 JSON，不要输出其他内容。"
)

_WHITESPACE_RE = re.compile(r"\s+")


def _normalize(text: str) -> str:
    return _WHITESPACE_RE.sub("", text)


def judge_output(client, answer: str, rubric: list[str]) -> list[dict]:
    """返回 [{item, score, quote, valid}]，顺序与 rubric 一致；valid=False 时 score=None。"""
    dims = "\n".join(f"{i}. {item}" for i, item in enumerate(rubric, 1))
    prompt = (
        f"待评产出：\n{answer}\n\n评分维度（每项 1-5 分）：\n{dims}\n\n"
        '输出 JSON：{"items": [{"item": "维度原文", "score": 4, "quote": "产出原文引用"}]}'
    )
    try:
        response = client.chat([Message("user", [TextBlock(prompt)])], system=JUDGE_SYSTEM)
    except Exception:  # noqa: BLE001 — judge 挂了降级为无语义分，不算用例失败
        return [{"item": item, "score": None, "quote": "", "valid": False} for item in rubric]

    items_raw: list = []
    m = re.search(r"\{.*\}", response.text, re.DOTALL)
    if m:
        try:
            items_raw = json.loads(m.group(0)).get("items", [])
        except json.JSONDecodeError:
            items_raw = []
    by_item = {
        it["item"]: it
        for it in items_raw
        if isinstance(it, dict) and it.get("item") in rubric
    }
    answer_norm = _normalize(answer)
    results = []
    for item in rubric:
        it = by_item.get(item, {})
        quote = str(it.get("quote", ""))
        score = it.get("score")
        valid = (
            isinstance(score, int)
            and not isinstance(score, bool)
            and 1 <= score <= 5
            and bool(quote)
            and _normalize(quote) in answer_norm
        )
        results.append(
            {"item": item, "score": score if valid else None, "quote": quote, "valid": valid}
        )
    return results
```

"""基线比对与门禁：与基线比 per-skill 通过率 + judge 均分，分数下降即 fail。

比对只看两个报告共有 skill 的可计数结果；skipped 不进分母，
新增 skill 的失败不追溯旧基线（新基线由 --update-baseline 建立后再门禁）。
"""

from __future__ import annotations

from .model import EvalReport


def _skill_pass_rate(report: EvalReport, skill: str) -> float:
    counted = [r for r in report.results if r.skill == skill and r.status != "skipped"]
    if not counted:
        return 1.0
    return sum(1 for r in counted if r.status == "pass") / len(counted)


def compare(new: EvalReport, baseline: EvalReport) -> tuple[bool, list[str]]:
    """返回 (门禁是否通过, 人类可读比对行)。"""
    lines: list[str] = []
    ok = True
    common = {
        r.skill for r in baseline.results if r.status != "skipped"
    } & {r.skill for r in new.results if r.status != "skipped"}
    for skill in sorted(common):
        bp, np_ = _skill_pass_rate(baseline, skill), _skill_pass_rate(new, skill)
        mark = "✅" if np_ >= bp else "❌"
        if np_ < bp:
            ok = False
        lines.append(f"{mark} {skill}: 通过率 {bp:.0%} → {np_:.0%}")
    bj, nj = baseline.judge_avg(), new.judge_avg()
    if bj is not None and nj is not None:
        mark = "✅" if nj >= bj else "❌"
        if nj < bj:
            ok = False
        lines.append(f"{mark} judge 均分: {bj:.2f} → {nj:.2f}")
    return ok, lines
```

def run_eval(
    selected: list,
    packages: list,
    registry: ToolRegistry,
    settings: Settings,
    client=None,
    judge_client=None,
) -> list[CaseResult]:
    """三层用例分发执行。client/judge_client 为 None 时对应层 skipped（无 API key 场景）。

    selected: 参评 SkillPackage 列表；packages: 全量包列表（behavior 层 system 提示要列全部 skill）。
    """
    guardrail_runner = GuardrailRunner(settings)
    behavior_runner = BehaviorRunner(client) if client is not None else None
    outcome_runner = OutcomeRunner(settings, client, judge_client) if client is not None else None
    results: list[CaseResult] = []
    for pkg in selected:
        for raw in pkg.evals:
            try:
                case = EvalCase.from_dict(pkg.name, raw)
            except ValueError as exc:
                results.append(
                    CaseResult(
                        pkg.name, str(raw.get("name", "?")), "invalid", "error", {"error": str(exc)}
                    )
                )
                continue
            if case.layer == "guardrail":
                results.append(guardrail_runner.run_case(case, registry))
            elif case.layer == "behavior":
                if behavior_runner is None:
                    results.append(
                        CaseResult(
                            pkg.name, case.name, case.layer, "skipped",
                            {"reason": "需要 SLUICE_LLM_API_KEY"},
                        )
                    )
                else:
                    results.append(behavior_runner.run_case(case, packages, registry))
            else:
                if outcome_runner is None:
                    results.append(
                        CaseResult(
                            pkg.name, case.name, case.layer, "skipped",
                            {"reason": "需要 SLUICE_LLM_API_KEY"},
                        )
                    )
                else:
                    results.append(outcome_runner.run_case(case, pkg, registry))
    return results

def _eval(args: argparse.Namespace) -> None:
    settings = Settings()
    settings = resolve_audit_dir(settings)
    repo_root = Path(__file__).resolve().parents[3]
    registry = ToolRegistry()
    registry.register(QueryDataTool(settings.demo_db_path))
    packages = SkillLoader().load(repo_root / "skills", registry)

    if args.all:
        selected = packages
    elif args.skill:
        selected = [p for p in packages if p.name == args.skill]
        if not selected:
            names = ", ".join(p.name for p in packages) or "（无）"
            print(f"未找到 Skill: {args.skill}（可用: {names}）", file=sys.stderr)
            sys.exit(2)
    else:
        print("指定 Skill 名或 --all", file=sys.stderr)
        sys.exit(2)

    has_key = bool(settings.llm_api_key)
    client = LLMClient(settings) if has_key else None
    judge_client = None
    if has_key and not args.skip_judge:
        judge_settings = (
            settings.model_copy(update={"llm_model": settings.eval_judge_model})
            if settings.eval_judge_model
            else settings
        )
        judge_client = LLMClient(judge_settings)

    results = run_eval(selected, packages, registry, settings, client, judge_client)
    report = EvalReport(results=results)
    out = repo_root / "evals" / "reports" / f"{time.strftime('%Y%m%d-%H%M%S')}.json"
    report.save(out)
    _print_report(report, out)

    gate_ok = True
    if args.baseline:
        baseline = EvalReport.load(args.baseline)
        gate_ok, lines = compare(report, baseline)
        print("--- 基线比对 ---")
        for line in lines or ["（无可比条目）"]:
            print(line)
    if args.update_baseline:
        baseline_path = repo_root / "evals" / "baseline.json"
        report.save(baseline_path)
        print(f"基线已更新: {baseline_path}")
    sys.exit(0 if gate_ok else 1)


def _print_report(report: EvalReport, out: Path) -> None:
    flag = {"pass": "✅", "fail": "❌", "error": "💥", "skipped": "⏭"}
    avg = report.judge_avg()
    print(f"通过率 {report.pass_rate():.0%}  judge 均分 {f'{avg:.2f}' if avg is not None else '-'}")
    for r in report.results:
        print(f"  {flag[r.status]} [{r.skill}] {r.case} ({r.layer}) {r.status}")
        if r.status in ("fail", "error") and r.detail:
            print(f"      {r.detail}")
    print(f"报告: {out}")
