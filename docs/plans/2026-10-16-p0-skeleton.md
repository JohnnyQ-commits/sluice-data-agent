```python
"""Anthropic Messages API 形状的消息/工具类型（自实现 function calling 的协议载体）。"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Any


@dataclass
class TextBlock:
    text: str


@dataclass
class ToolUseBlock:
    id: str
    name: str
    input: dict[str, Any]


@dataclass
class ToolResultBlock:
    tool_use_id: str
    content: str
    is_error: bool = False


@dataclass
class ToolDef:
    """工具定义——JSON Schema 进 API 的 input_schema 字段。"""

    name: str
    description: str
    parameters: dict[str, Any]

    def to_api(self) -> dict[str, Any]:
        return {
            "name": self.name,
            "description": self.description,
            "input_schema": self.parameters,
        }


@dataclass
class Message:
    """一条消息。content 是块列表（user 可含 tool_result，assistant 可含 tool_use）。"""

    role: str  # "user" | "assistant"
    content: list  # list[TextBlock | ToolUseBlock | ToolResultBlock]


@dataclass
class LLMResponse:
    content: list  # list[TextBlock | ToolUseBlock]
    stop_reason: str | None  # "tool_use" | "end_turn" | ...
    usage: dict[str, Any]

    @property
    def text(self) -> str:
        return "".join(b.text for b in self.content if isinstance(b, TextBlock))

    @property
    def tool_uses(self) -> list[ToolUseBlock]:
        return [b for b in self.content if isinstance(b, ToolUseBlock)]
```

```python
"""LLM 网关客户端——直接调 Anthropic Messages API（httpx），自实现 function calling。

重试策略：超时与 429/5xx 可重试，指数退避；4xx 客户端错误不重试直接抛。
"""
from __future__ import annotations

import time
from typing import Any

import httpx

from ..config.settings import Settings
from ..exceptions import LLMHTTPError, LLMTimeoutError
from .types import LLMResponse, Message, TextBlock, ToolDef, ToolUseBlock

_RETRYABLE_STATUS = {429, 500, 502, 503, 529}


class LLMClient:
    def __init__(self, settings: Settings):
        self._settings = settings

    def chat(
        self,
        messages: list[Message],
        system: str | None = None,
        tools: list[ToolDef] | None = None,
    ) -> LLMResponse:
        payload: dict[str, Any] = {
            "model": self._settings.llm_model,
            "max_tokens": 4096,
            "messages": [self._encode_message(m) for m in messages],
        }
        if system:
            payload["system"] = system
        if tools:
            payload["tools"] = [t.to_api() for t in tools]

        headers = {
            "x-api-key": self._settings.llm_api_key,
            "anthropic-version": "2023-06-01",
            "content-type": "application/json",
        }

        last_exc: Exception | None = None
        for attempt in range(self._settings.llm_max_retries + 1):
            try:
                resp = httpx.post(
                    f"{self._settings.llm_base_url}/v1/messages",
                    headers=headers,
                    json=payload,
                    timeout=self._settings.llm_timeout_seconds,
                )
            except httpx.TimeoutException as exc:
                last_exc = exc
            else:
                if resp.status_code == 200:
                    return self._parse_response(resp.json())
                err = LLMHTTPError(resp.status_code, resp.text)
                if resp.status_code in _RETRYABLE_STATUS:
                    last_exc = err
                else:
                    raise err
            if attempt < self._settings.llm_max_retries:
                time.sleep(2**attempt)

        if isinstance(last_exc, httpx.TimeoutException) or last_exc is None:
            raise LLMTimeoutError(
                f"LLM 请求超时（重试 {self._settings.llm_max_retries} 次后放弃）"
            ) from last_exc
        raise last_exc

    @staticmethod
    def _encode_message(m: Message) -> dict[str, Any]:
        blocks: list[dict[str, Any]] = []
        for b in m.content:
            if isinstance(b, TextBlock):
                blocks.append({"type": "text", "text": b.text})
            elif isinstance(b, ToolUseBlock):
                blocks.append({"type": "tool_use", "id": b.id, "name": b.name, "input": b.input})
            else:  # ToolResultBlock
                blocks.append(
                    {
                        "type": "tool_result",
                        "tool_use_id": b.tool_use_id,
                        "content": b.content,
                        **({"is_error": True} if b.is_error else {}),
                    }
                )
        return {"role": m.role, "content": blocks}

    @staticmethod
    def _parse_response(data: dict[str, Any]) -> LLMResponse:
        content = []
        for b in data.get("content", []):
            if b.get("type") == "text":
                content.append(TextBlock(b["text"]))
            elif b.get("type") == "tool_use":
                content.append(ToolUseBlock(b["id"], b["name"], b.get("input", {})))
        return LLMResponse(
            content=content,
            stop_reason=data.get("stop_reason"),
            usage=data.get("usage", {}),
        )
```

```python
"""统一工具协议——内置工具、Skill 私有工具、MCP 外部工具都归一到这里。"""
from __future__ import annotations

from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Any

from ..llm.types import ToolDef


@dataclass
class ToolResult:
    content: str
    is_error: bool = False


class Tool(ABC):
    """工具基类。子类必须定义 name/description/parameters 并实现 execute。"""

    name: str
    description: str
    parameters: dict[str, Any]  # JSON Schema

    @abstractmethod
    def execute(self, args: dict[str, Any]) -> ToolResult: ...

    def to_def(self) -> ToolDef:
        return ToolDef(self.name, self.description, self.parameters)
```

`src/sluice/tools/registry.py`:

```python
from __future__ import annotations

from .base import Tool
from ..llm.types import ToolDef


class ToolRegistry:
    """工具池。模型视角只有一个扁平工具池，来源（内置/Skill/MCP）无关。"""

    def __init__(self):
        self._tools: dict[str, Tool] = {}

    def register(self, tool: Tool) -> None:
        if tool.name in self._tools:
            raise ValueError(f"工具重复注册: {tool.name}")
        self._tools[tool.name] = tool

    def get(self, name: str) -> Tool | None:
        return self._tools.get(name)

    def to_defs(self) -> list[ToolDef]:
        return [t.to_def() for t in self._tools.values()]
```

```python
"""ReAct 主循环：Thought → Action(工具调用) → Observation → … → Final Answer。

这是整个平台的发动机。guardrails 在 P2 接入本循环的三个拦截点
（工具调用前 / 循环运行中 / 产出交付前），P0 只保留轮次上限和未知工具回喂。
"""
from __future__ import annotations

from ..exceptions import RoundsExhaustedError
from ..llm.client import LLMClient
from ..llm.types import LLMResponse, Message, TextBlock, ToolResultBlock, ToolUseBlock
from ..tools.registry import ToolRegistry

DEFAULT_SYSTEM = "你是 Sluice 平台的助手。可以调用提供的工具完成任务，完成后直接给出最终回答。"


class ReActRuntime:
    def __init__(
        self,
        client: LLMClient,
        registry: ToolRegistry,
        system: str = DEFAULT_SYSTEM,
        max_rounds: int = 10,
    ):
        self._client = client
        self._registry = registry
        self._system = system
        self._max_rounds = max_rounds
        self.trace: list[dict] = []  # 每轮轨迹，P1 经 SSE 推给前端

    def run(self, user_input: str) -> str:
        self.trace = []
        messages: list[Message] = [Message("user", [TextBlock(user_input)])]

        for round_no in range(1, self._max_rounds + 1):
            response = self._client.chat(
                messages, system=self._system, tools=self._registry.to_defs()
            )
            tool_uses = response.tool_uses

            if not tool_uses:  # Final Answer
                self._record(round_no, response, [])
                return response.text

            messages.append(Message("assistant", response.content))
            observations = []
            for tu in tool_uses:
                observations.append(self._execute_tool(tu))
            messages.append(Message("user", observations))
            self._record(round_no, response, observations)

        raise RoundsExhaustedError(f"ReAct 循环超过 {self._max_rounds} 轮未产出最终回答")

    def _execute_tool(self, tu: ToolUseBlock) -> ToolResultBlock:
        tool = self._registry.get(tu.name)
        if tool is None:
            return ToolResultBlock(tu.id, f"未知工具: {tu.name}", is_error=True)
        try:
            result = tool.execute(tu.input)
            return ToolResultBlock(tu.id, result.content, is_error=result.is_error)
        except Exception as exc:  # noqa: BLE001 — 工具异常回喂模型自纠
            return ToolResultBlock(tu.id, f"工具执行异常: {exc}", is_error=True)

    def _record(self, round_no: int, response: LLMResponse, observations: list) -> None:
        self.trace.append(
            {
                "round": round_no,
                "thought": response.text,
                "tool_calls": [
                    {
                        "name": tu.name,
                        "args": tu.input,
                        "observation": obs.content,
                        "is_error": obs.is_error,
                    }
                    for tu, obs in zip(response.tool_uses, observations)
                ],
            }
        )
```

```python
"""sluice CLI。P0 只有一个子命令：run（单次指令跑完整 ReAct 循环）。"""
from __future__ import annotations

import argparse
import sys

from ..config.settings import Settings
from ..exceptions import SluiceError
from ..llm.client import LLMClient
from ..runtime.loop import ReActRuntime
from ..tools.echo import EchoTool
from ..tools.registry import ToolRegistry


def main(argv: list[str] | None = None) -> None:
    parser = argparse.ArgumentParser(
        prog="sluice", description="Sluice — 可挂载 Skill 的数据 Agent 平台"
    )
    sub = parser.add_subparsers(dest="command", required=True)
    run_p = sub.add_parser("run", help="单次指令：跑完整 ReAct 循环并打印轨迹")
    run_p.add_argument("message", help="发给 Agent 的指令")
    args = parser.parse_args(argv)

    if args.command == "run":
        _run(args.message)


def _run(message: str) -> None:
    settings = Settings()
    if not settings.llm_api_key:
        print("缺少 SLUICE_LLM_API_KEY（复制 .env.example 为 .env 并填写）", file=sys.stderr)
        sys.exit(1)

    registry = ToolRegistry()
    registry.register(EchoTool())
    runtime = ReActRuntime(
        LLMClient(settings), registry, max_rounds=settings.react_max_rounds
    )

    print(f"指令: {message}\n---")
    try:
        final = runtime.run(message)
    except SluiceError as exc:
        print(f"[失败] {exc}", file=sys.stderr)
        sys.exit(1)

    for event in runtime.trace:
        print(f"第 {event['round']} 轮")
        if event["thought"]:
            print(f"  Thought: {event['thought']}")
        for tc in event["tool_calls"]:
            status = "错误" if tc["is_error"] else "观测"
            print(f"  Action: {tc['name']}({tc['args']})")
            print(f"  {status}: {tc['observation']}")
    print("---")
    print(f"最终回答: {final}")


if __name__ == "__main__":
    main()
```
